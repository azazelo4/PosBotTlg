flowchart TD
    A[Запуск бота start] --> B[Отправка запроса на предоставление контакта]
    B --> C[Пользователь делится контактом]
    C --> D[Авторизация пользователя]
    D --> E{Роль пользователя?}
    E -- cashier --> F[Отправка кнопок для кассира]
    E -- admin --> G[Отправка кнопок для администратора]
    F --> H[Команда Касса]
    G --> H
    H --> I[Обработка кассовых операций]
    I --> J{Выбор операции}
    J -- Продажа --> K[Запуск процесса продажи]
    K --> L[Ввод артикула]
    L --> M[Поиск товара в БД]
    M --> N[Ввод суммы продажи]
    N --> O[Проверка минимальной цены]
    O --> P[Выбор типа оплаты]
    P --> Q[Подтверждение продажи]
    Q --> R[Запись продажи в БД]
    J -- Другие операции --> S[Обработка других кассовых действий]
    E -- иные роли --> T[Ограниченный доступ]
